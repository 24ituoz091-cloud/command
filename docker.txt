Create src\Main.java
@'
import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.PreparedStatement;
import java.sql.ResultSet;

public class Main {
    public static void main(String[] args) throws Exception {
        String url = "jdbc:mysql://db:3306/studentdb";
        String user = "student";
        String password = System.getenv("DB_PASSWORD");

        Connection con = null;

        // Wait for MySQL to become available.
        for (int i = 0; i < 30; i++) {
            try {
                con = DriverManager.getConnection(url, user, password);
                break;
            } catch (Exception e) {
                System.out.println("Waiting for MySQL...");
                Thread.sleep(2000);
            }
        }

        if (con == null) {
            throw new RuntimeException("Could not connect to MySQL");
        }

        try (Connection connection = con) {
            String createTable =
                "CREATE TABLE IF NOT EXISTS students (" +
                "id INT PRIMARY KEY, name VARCHAR(100))";

            try (PreparedStatement ps =
                     connection.prepareStatement(createTable)) {
                ps.executeUpdate();
            }

            // INSERT operation
            String insert =
                "INSERT INTO students (id, name) VALUES (?, ?) " +
                "ON DUPLICATE KEY UPDATE name = VALUES(name)";

            try (PreparedStatement ps =
                     connection.prepareStatement(insert)) {
                ps.setInt(1, 1);
                ps.setString(2, "Aksh");
                ps.executeUpdate();
            }

            System.out.println("INSERT successful");

            // UPDATE operation
            String update =
                "UPDATE students SET name = ? WHERE id = ?";

            try (PreparedStatement ps =
                     connection.prepareStatement(update)) {
                ps.setString(1, "Aksh Updated");
                ps.setInt(2, 1);
                ps.executeUpdate();
            }

            System.out.println("UPDATE successful");

            // Display the final record
            try (PreparedStatement ps =
                     connection.prepareStatement(
                         "SELECT id, name FROM students");
                 ResultSet rs = ps.executeQuery()) {
                while (rs.next()) {
                    System.out.println(
                        rs.getInt("id") + " : " + rs.getString("name"));
                }
            }
        }
    }
}
'@ | Set-Content .\src\Main.java
